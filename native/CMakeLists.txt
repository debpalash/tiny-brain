cmake_minimum_required(VERSION 3.20)
project(snn_native CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
set(CMAKE_CXX_FLAGS_RELEASE "-O3")
find_package(Threads REQUIRED)
enable_testing()

# Backend-neutral core: network, parameters, CPU reference backend. Plain C++20.
add_library(snn_core STATIC src/network_builder.cpp src/cpu_backend.cpp)
target_include_directories(snn_core PUBLIC src)
target_link_libraries(snn_core PUBLIC Threads::Threads)

set(SNN_LIBS snn_core)
set(SNN_DEFS "")
if(APPLE)
  add_library(snn_metal STATIC src/metal_backend.cpp)
  target_include_directories(snn_metal PUBLIC src third_party/metal-cpp)
  target_compile_definitions(snn_metal PRIVATE SNN_KERNEL_DIR="${CMAKE_CURRENT_SOURCE_DIR}/kernels")
  target_link_libraries(snn_metal PUBLIC snn_core "-framework Foundation" "-framework Metal" "-framework QuartzCore")
  list(APPEND SNN_LIBS snn_metal)
  set(SNN_DEFS SNN_HAVE_METAL=1)
endif()

add_executable(snn_test tests/test_equivalence.cpp)
target_link_libraries(snn_test PRIVATE ${SNN_LIBS})
target_compile_definitions(snn_test PRIVATE ${SNN_DEFS})
add_test(NAME cpu_vs_metal COMMAND snn_test)

add_executable(snn_bench bench/bench.cpp)
target_link_libraries(snn_bench PRIVATE ${SNN_LIBS})
target_compile_definitions(snn_bench PRIVATE ${SNN_DEFS} SNN_RESULTS_DIR="${CMAKE_CURRENT_SOURCE_DIR}/results")

# ---- Larva brain (CPU): connectome loader, live simulation, learning rule, WebSocket viewer server.
add_library(larva_core STATIC src/larva_data.cpp src/larva_brain.cpp src/larva_experiment.cpp)
target_include_directories(larva_core PUBLIC src)
target_link_libraries(larva_core PUBLIC snn_core)
set(LARVA_DATA_DIR "${CMAKE_CURRENT_SOURCE_DIR}/data/larva")
target_compile_definitions(larva_core PUBLIC LARVA_DATA_DIR="${LARVA_DATA_DIR}")
add_executable(larva_experiment tests/larva_experiment_main.cpp)
target_link_libraries(larva_experiment PRIVATE larva_core)
add_executable(larva_server src/larva_server.cpp src/ws_server.cpp)
target_link_libraries(larva_server PRIVATE larva_core)

add_executable(larva_test tests/larva_test.cpp)
target_link_libraries(larva_test PRIVATE larva_core)
add_test(NAME larva_loader COMMAND larva_test loader)
add_test(NAME larva_stability COMMAND larva_test stability)
add_test(NAME larva_learning COMMAND larva_test learning)
set_tests_properties(larva_learning PROPERTIES TIMEOUT 900)
find_program(LARVA_PYTHON python3)
if(LARVA_PYTHON)
  add_test(NAME larva_protocol COMMAND ${LARVA_PYTHON} -I ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_protocol.py $<TARGET_FILE:larva_server>)
  set_tests_properties(larva_protocol PROPERTIES TIMEOUT 300)
endif()
